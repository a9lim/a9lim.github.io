\section*{Selected Research}

\entry{Saklas}{https://github.com/a9lim/saklas}{Mechanistic interpretability workbench - Python, PyTorch}{Apr 2026 - Present}
\begin{itemize}
  \item Built a local workbench for probing and steering LLMs along curved manifolds.
  \item Implemented Jacobian-lens readouts and arbitrary-role chat support.
  \item Published a typed Python API, dashboard, and OpenAI/Ollama-compatible server compatible with most
  open source LLMs. Available on PyPI with CUDA and MPS support.
\end{itemize}

\entry{Introspection via Kaomoji}{https://a9l.im/blog/introspection-via-kaomoji}{Emotional concept representation study - Python}{Apr 2026 - Jun 2026}
\begin{itemize}
  \item The emotional category of a prompt was decodable with 99.2\% accuracy from the hidden state and the
  opening kaomoji alone retained 80.6\% accuracy on Gemma against 11.1\% by chance.
  \item Procrustes-aligned emotional category centroids to recover shared geometry; built kaomoji
  collection tooling and published the collected corpus.
\end{itemize}

\entry{Elapsed-Time Encoding in LLMs}{https://github.com/a9lim/time-experiment}{Ongoing time encoding study - Python, PyTorch}{Jun 2026 - Present}
\begin{itemize}
  \item Showed elapsed conversational time is linearly encoded at an elicited token ($r=0.88$, about 0.3
  seconds per token) using linear probes.
\end{itemize}

\entry{Neuralese as a Conlang}{https://github.com/a9lim/acot-experiment}{Ongoing abstract-CoT analysis study - Python, PyTorch}{Jul 2026 - Present}
\begin{itemize}
  \item Implemented Abstract-CoT to test if abstract tokens could be interpreted as a compositional language.
  \item A preliminary 135M run did not successfully learn to use abstract tokens to reason.
\end{itemize}

\entry{Block Sparse Crosscoders}{https://github.com/a9lim/block-crosscoder-experiment}{Ongoing cross-layer representation geometry study - Python, PyTorch}{Jul 2026 - Present}
\begin{itemize}
  \item Synthesized recent work on block sparse autoencoders and sparse crosscoders to extract cross-layer multidimensional features.
\end{itemize}
