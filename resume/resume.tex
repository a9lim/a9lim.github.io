\section*{Selected Research}

\entry{Saklas}{https://github.com/a9lim/saklas}{Mechanistic interpretability workbench - Python, PyTorch}{Apr 2026 - Present}
\begin{itemize}
  \item Built a local workbench for extracting, probing, and steering LLMs with behavioral manifolds, with 
  Mahalanobis whitening, support for Jacobian-lens readouts, and role-agnostic chat templates.
  \item Published a typed Python API, dashboard, and OpenAI/Ollama-compatible server tested on Gemma,
  Qwen, Mistral, Llama, GLM, gpt-oss, and Talkie. Available on PyPI with CUDA and MPS support.
\end{itemize}

\entry{Introspection via Kaomoji}{https://a9l.im/blog/introspection-via-kaomoji}{Representation study and Llmoji corpus - Python, Hugging Face}{Apr 2026 - Present}
\begin{itemize}
  \item Across five open-weight model families, classifying the opening kaomoji could predict the emotional 
  content of a prompt with 80.6\% accuracy on Gemma versus 11.1\% by chance.
  \item Procrustes-aligned emotional category centroids to recover shared geometry; built kaomoji 
  collection tool and published the corpus.
\end{itemize}

\entry{Elapsed-Time Encoding in LLMs}{https://github.com/a9lim/time-experiment}{Residual-stream probing study - Python, PyTorch}{Jun 2026 - Present}
\begin{itemize}
  \item Showed elapsed conversational time is linearly encoded at an elicited token ($r=0.88$, about 0.3
  seconds per token) by using linear probes.
\end{itemize}

\entry{Neuralese as a Conlang}{https://github.com/a9lim/acot-experiment}{Ongoing Abstract-CoT analysis study - Python, PyTorch}{Jul 2026 - Present}
\begin{itemize}
  \item Implemented Abstract-CoT to test if a reasoning code could be interpreted as a compositional language.
  \item A preliminary 135M run did not successfully learn to use abstract tokens to reason.
\end{itemize}

\entry{Block Sparse Crosscoders}{https://github.com/a9lim/block-crosscoder-experiment}{Ongoing cross-layer neural geometry study - Python, PyTorch}{Jul 2026 - Present}
\begin{itemize}
  \item Synthesized recent work on block sparse autoencoders and sparse crosscoders to extract cross-layer multidimensional features.
\end{itemize}
